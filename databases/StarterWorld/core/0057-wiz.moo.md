## Verb: @corify

```yaml
names: ["@corify"]
dobj: any
prep: as
iobj: any
owner: "#0"
flags:
  - readable
  - executable
```

```csharp
if (!player.wizard)
  player:notify("Sorry.");
  return;
endif

obj = dobj;
name = $string_utils:trim(iobjstr);

if (!valid(obj))
  player:tell("You don't see that here.");
  return;
endif

if (!name)
  player:tell("Usage: @corify <object> as <name>");
  return;
endif

if (name[1] == "$")
  name = name[2..$];
endif

if (!match(name, "^[A-Za-z_][A-Za-z0-9_]*$"))
  player:tell("Core names must look like MOO identifiers.");
  return;
endif

if ($object_utils:has_property(#0, name))
  player:tell(tostr("$", name, " already exists."));
  return;
endif

set_name(obj, "$" + name);
move(obj, $nothing);
obj.r = 1;
obj.f = 1;

if ($object_utils:has_property(obj, "_db"))
  obj._db = "core";
else
  add_property(obj, "_db", "core", {player, "rw"});
endif

add_property(#0, name, obj, {player, "rc"});
player:tell(tostr(obj.name, " is now corified as $", name, "."));
```

## Verb: @shutdown

```yaml
names: ["@shutdown"]
dobj: any
prep: any
iobj: any
owner: "#0"
flags:
  - readable
  - executable
```

```csharp
if (!player.wizard)
  player:notify("Sorry.");
  return;
endif

in_progress = 0;
if (typeof($shutdown_task) == INT)
  for task in (queued_tasks())
    if (task[1] == $shutdown_task)
      in_progress = 1;
    endif
  endfor
endif

if (in_progress)
  player:notify(tostr("Shutdown already in progress. The MOO will be shut down in ", $shutdown_time - time(), " seconds, by ", $shutdown_message));
  return;
endif

if (argstr == "now" || index(argstr, "now ") == 1)
  delay = 0;
  argstr = length(argstr) > 3 ? argstr[5..$] | "";
elseif (s = match(argstr, "^in +%([0-9]+%)%( +%|$%)"))
  bounds = s[3][1];
  delay = toint(argstr[bounds[1]..bounds[2]]);
  argstr = argstr[s[2] + 1..$];
else
  delay = 2;
endif

$shutdown_time = time() + delay * 60;
$shutdown_message = tostr(player.name, " (", player, "): ", argstr);

if (delay <= 0)
  notify(player, tostr("*** Server shutdown by ", player.name, " (", player, "): ", argstr, " ***"));
  $shutdown_task = E_NONE;
  shutdown("");
  return;
endif

notify(player, tostr("*** The server will be shut down by ", player.name, " (", player, ") in ", delay, " minute", delay == 1 ? "" | "s", argstr ? tostr(": ", argstr) | "", " ***"));

fork shutdown_task (delay * 60)
  notify(player, tostr("*** Server shutdown by ", $shutdown_message, " ***"));
  $shutdown_task = E_NONE;
  shutdown("");
endfork

$shutdown_task = shutdown_task;
```
